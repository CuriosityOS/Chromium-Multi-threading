void SharedGpuContext::PostToMainThread(CrossThreadOnceClosure task) {
  if (main_thread_task_poster_) {
    main_thread_task_poster_.Run(std::move(task));
    return;
  }
  PostCrossThreadTask(
      *Thread::MainThread()->GetTaskRunner(MainThreadTaskRunnerRestricted()),
      FROM_HERE, std::move(task));
}

// static
void SharedGpuContext::SetMainThreadTaskPosterForCurrentThread(
    MainThreadTaskPoster poster) {
  GetInstanceForCurrentThread()->main_thread_task_poster_ = std::move(poster);
}
