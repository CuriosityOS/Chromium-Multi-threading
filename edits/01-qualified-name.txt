static scoped_refptr<QualifiedNameImpl> LookupOrAddQualifiedName(
    const QualifiedNameData& data) {
  scoped_refptr<QualifiedNameImpl> result;
  bool is_new_entry = false;
  {
    base::AutoLock locker(QualifiedNameCacheLock());
    QualifiedNameCache::AddResult add_result =
        GetQualifiedNameCache().AddWithTranslator<QNameComponentsTranslator>(
            data);
    // Take our reference while holding the lock so that a concurrent
    // ReleaseSlow() on another thread cannot destroy the impl under us.
    result = *add_result.stored_value;
    is_new_entry = add_result.is_new_entry;
  }
  if (is_new_entry) {
    // Drop the initial reference created by QNameComponentsTranslator::Store;
    // the count is >= 2 here so this never takes the slow path.
    result->Release();
  }
  return result;
}

void QualifiedNameImpl::ReleaseSlow() const {
  base::AutoLock locker(QualifiedNameCacheLock());
  if (ref_count_.fetch_sub(1, std::memory_order_acq_rel) != 1) {
    return;
  }
  GetQualifiedNameCache().erase(const_cast<QualifiedNameImpl*>(this));
  delete this;
}
